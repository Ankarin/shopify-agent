<script>
  (function () {
    const iframe = document.getElementById('chat-widget-iframe');
    
    if (!iframe) {
      console.error('[Chat Widget] iframe not found');
      return;
    }

    console.log('[Chat Widget] Initializing...', {
      userAgent: navigator.userAgent,
      screenWidth: window.innerWidth,
      isMobile: window.innerWidth <= 768
    });

    iframe.style.visibility = 'visible';
    iframe.style.display = 'block';
    iframe.style.opacity = '1';

    function isMobile() {
      return window.innerWidth <= 768;
    }

    function setIframeStyle(styles) {
      requestAnimationFrame(function() {
        Object.keys(styles).forEach(function(key) {
          iframe.style.setProperty(key, styles[key], 'important');
        });
      });
    }

    function updateIframePosition() {
      const mobile = isMobile();
      const isOpen = iframe.dataset.isOpen === 'true';
      
      if (mobile && isOpen) {
        setIframeStyle({
          bottom: '0',
          right: '0',
          top: '0',
          left: '0',
          width: '100%',
          height: '100%',
          maxWidth: '100%',
          maxHeight: '100%'
        });
      } else if (!mobile && isOpen) {
        setIframeStyle({
          bottom: '20px',
          right: '20px',
          top: 'auto',
          left: 'auto',
          width: '500px',
          height: '780px',
          maxWidth: 'calc(100vw - 40px)',
          maxHeight: 'calc(100vh - 40px)'
        });
      } else {
        setIframeStyle({
          bottom: '20px',
          right: '20px',
          top: 'auto',
          left: 'auto',
          width: '100px',
          height: '100px',
          maxWidth: '100px',
          maxHeight: '100px'
        });
      }
    }

    iframe.addEventListener('load', function() {
      console.log('[Chat Widget] iframe loaded successfully');
      iframe.style.visibility = 'visible';
      iframe.style.opacity = '1';
      updateIframePosition();
      
      setTimeout(function() {
        var rect = iframe.getBoundingClientRect();
        console.log('[Chat Widget] iframe dimensions:', {
          width: rect.width,
          height: rect.height,
          bottom: rect.bottom,
          right: rect.right,
          visible: rect.width > 0 && rect.height > 0
        });
      }, 500);
    });

    window.addEventListener('message', function (event) {
      if (event.data && event.data.type === 'widget-resize') {
        const { isOpen } = event.data;
        iframe.dataset.isOpen = isOpen;
        updateIframePosition();
      }
    });

    window.addEventListener('resize', updateIframePosition);
    window.addEventListener('orientationchange', updateIframePosition);

    updateIframePosition();
    
    setTimeout(function() {
      iframe.style.visibility = 'visible';
      iframe.style.opacity = '1';
    }, 100);
  })();
</script>
